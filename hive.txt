5.Demonstrate Order by and Group by clause in Hive Environment
------------------------------------------------------------------------


CREATE TABLE emp (id INT, name STRING, salary FLOAT) ROW FORMAT DELIMITED FIELDS TERMINATED BY ','; 
LOAD DATA LOCAL INPATH '/home/hadoop/emp.txt' INTO TABLE emp;
SELECT * FROM emp ORDER BY salary; 
SELECT * FROM emp ORDER BY salary DESC; 
SELECT dept, SUM(salary) FROM emp GROUP BY dept;
